    80→function ToggleRow({
    81→  icon: Icon,
    82→  title,
    83→  desc,
    84→  checked,
    85→  onToggle,
    86→  disabled,
    87→  accent = "primary",
    88→}: {
    89→  icon: React.ElementType;
    90→  title: string;
    91→  desc: string;
    92→  checked: boolean;
    93→  onToggle: (v: boolean) => void;
    94→  disabled?: boolean;
    95→  accent?: "primary" | "red" | "amber";
    96→}) {
    97→  const accentClass =
    98→    accent === "red"
    99→      ? "text-red-500 bg-red-500/10"
   100→      : accent === "amber"
   101→        ? "text-amber-500 bg-amber-500/10"
   102→        : "text-primary bg-primary/10";
   103→  return (
   104→    <div
   105→      data-checked={checked ? "true" : "false"}
   106→      className={cn(
   107→        "toggle-glow flex items-center justify-between gap-3 rounded-xl border p-3 transition-all duration-200",
   108→        checked
   109→          ? "border-primary/40 bg-primary/5"
   110→          : "border-border/50 bg-card/40 hover:border-border hover:bg-card/70",
   111→      )}
   112→    >
   113→      <div className="flex min-w-0 items-center gap-3">
   114→        <div
   115→          className={cn(
   116→            "grid size-9 shrink-0 place-items-center rounded-lg transition-transform duration-200",
   117→            accentClass,
   118→            checked && "scale-105",
   119→          )}
   120→        >
   121→          <Icon className="size-[18px]" strokeWidth={2.1} />
   122→        </div>
   123→        <div className="min-w-0">
   124→          <div className="text-sm font-medium leading-tight">{title}</div>
   125→          <div className="text-[11px] leading-snug text-muted-foreground">{desc}</div>
   126→        </div>
   127→      </div>
   128→      <Switch checked={checked} onCheckedChange={onToggle} disabled={disabled} aria-label={title} />
   129→    </div>
   130→  );
   131→}

   552→              onCheckedChange={(v) => rec.updateSettings("watermark", v)}
   553→              disabled={disabled}
   554→              aria-label={t("watermark")}
   555→            />
   556→          </label>
   557→          {settings.watermark && (
   558→            <div className="space-y-3 rounded-xl border border-border/50 bg-muted/30 p-3">
   559→              <div className="text-[11px] font-medium text-muted-foreground">{t("watermarkCustom")}</div>
   560→              <div className="space-y-1.5">
   561→                <Label className="text-[10px] text-muted-foreground">{t("watermarkText")}</Label>
   562→                <Input
   563→                  value={settings.watermarkText}
   564→                  onChange={(e) => rec.updateSettings("watermarkText", e.target.value)}
   565→                  placeholder={t("watermarkTextPlaceholder")}
   566→                  disabled={disabled}
   567→                  className="h-8 text-xs"
   568→                />
   569→                <Button
   570→                  variant="ghost"
   571→                  size="sm"
   572→                  className="h-6 gap-1 px-2 text-[10px] text-muted-foreground"
   573→                  onClick={() => rec.updateSettings("watermarkText", "")}
   574→                  disabled={disabled}
   575→                >
   576→                  {t("watermarkUseApp")}
   577→                </Button>
   578→              </div>
   579→              <div className="space-y-1.5">
   580→                <div className="flex items-center justify-between">
   581→                  <Label className="text-[10px] text-muted-foreground">{t("watermarkOpacity")}</Label>
   582→                  <span className="text-[10px] font-medium text-muted-foreground">
   583→                    {Math.round(settings.watermarkOpacity * 100)}%
   584→                  </span>
   585→                </div>
   586→                <Slider
   587→                  value={[settings.watermarkOpacity]}
   588→                  min={0.1}
   589→                  max={1}
   590→                  step={0.05}
   591→                  onValueChange={(v) => rec.updateSettings("watermarkOpacity", v[0])}
   592→                  disabled={disabled}
   593→                />
   594→              </div>
   595→              <div className="space-y-1.5">
   596→                <div className="flex items-center justify-between">
   597→                  <Label className="text-[10px] text-muted-foreground">{t("watermarkSize")}</Label>
   598→                  <span className="text-[10px] font-medium text-muted-foreground">
   599→                    {Math.round(settings.watermarkSize * 1000)}‰
   600→                  </span>
   601→                </div>
   602→                <Slider
   603→                  value={[settings.watermarkSize]}
   604→                  min={0.01}
   605→                  max={0.06}
   606→                  step={0.002}
   607→                  onValueChange={(v) => rec.updateSettings("watermarkSize", v[0])}
   608→                  disabled={disabled}
   609→                />
   610→              </div>
   611→
   612→              {/* Round 11: Watermark mode selector */}
   613→              <div className="space-y-1.5">
   614→                <Label className="text-[10px] text-muted-foreground">{t("watermarkMode")}</Label>
   615→                <Select
   616→                  value={settings.watermarkMode}
   617→                  onValueChange={(v) => rec.updateSettings("watermarkMode", v as "text" | "logo" | "both")}
   618→                  disabled={disabled}
   619→                >
   620→                  <SelectTrigger className="h-8 w-full text-xs">
   621→                    <SelectValue />
   622→                  </SelectTrigger>
   623→                  <SelectContent>
   624→                    <SelectItem value="text">{t("watermarkModeText")}</SelectItem>
   625→                    <SelectItem value="logo">{t("watermarkModeLogo")}</SelectItem>
   626→                    <SelectItem value="both">{t("watermarkModeBoth")}</SelectItem>
   627→                  </SelectContent>
   628→                </Select>
   629→              </div>
   630→
   631→              {/* Round 11: Logo upload (shown when mode includes logo) */}
   632→              {(settings.watermarkMode === "logo" || settings.watermarkMode === "both") && (
   633→                <div className="space-y-2 rounded-lg border border-border/50 bg-card/40 p-2.5">
   634→                  <Label className="text-[10px] text-muted-foreground">{t("watermarkLogo")}</Label>
   635→                  <p className="text-[9px] text-muted-foreground/70">{t("watermarkLogoDesc")}</p>
   636→                  <div className="flex items-center gap-2">
   637→                    <label className="flex h-8 cursor-pointer items-center gap-1.5 rounded-md border border-border bg-card px-2.5 text-xs font-medium transition-colors hover:bg-accent">
   638→                      <Upload className="size-3" />
   639→                      {t("watermarkLogoUpload")}
   640→                      <input
   641→                        type="file"
   642→                        accept="image/png,image/jpeg,image/svg+xml"
   643→                        className="hidden"
   644→                        onChange={(e) => {
   645→                          const file = e.target.files?.[0];
   646→                          if (!file) return;
   647→                          const reader = new FileReader();
   648→                          reader.onload = () => {
   649→                            const dataUrl = reader.result as string;
   650→                            rec.updateSettings("watermarkLogoDataUrl", dataUrl);
   651→                          };
   652→                          reader.onerror = () => {};
   653→                          reader.readAsDataURL(file);
   654→                          e.target.value = "";
   655→                        }}
   656→                      />
   657→                    </label>
   658→                    {settings.watermarkLogoDataUrl && (
   659→                      <Button
   660→                        variant="ghost"
   661→                        size="sm"
   662→                        className="h-8 gap-1 px-2 text-xs text-muted-foreground hover:text-destructive"
   663→                        onClick={() => rec.updateSettings("watermarkLogoDataUrl", "")}
   664→                        disabled={disabled}
   665→                      >
   666→                        <Trash2 className="size-3" />
   667→                        {t("watermarkLogoRemove")}
   668→                      </Button>
   669→                    )}
   670→                  </div>
   671→                  {/* Logo preview + size slider */}
   672→                  {settings.watermarkLogoDataUrl && (
   673→                    <div className="flex items-center gap-2">
   674→                      <div className="grid size-10 shrink-0 place-items-center overflow-hidden rounded border border-border/50 bg-black/40">
   675→                        <img src={settings.watermarkLogoDataUrl} alt="logo" className="max-h-full max-w-full object-contain" />
   676→                      </div>
   677→                      <div className="flex-1 space-y-1">
   678→                        <div className="flex items-center justify-between">
   679→                          <span className="text-[9px] text-muted-foreground">{t("watermarkLogoSize")}</span>
   680→                          <span className="text-[9px] font-medium text-muted-foreground">
   681→                            {Math.round(settings.watermarkLogoSize * 100)}%
   682→                          </span>
   683→                        </div>
   684→                        <Slider
   685→                          value={[settings.watermarkLogoSize]}
   686→                          min={0.02}
   687→                          max={0.15}
   688→                          step={0.005}
   689→                          onValueChange={(v) => rec.updateSettings("watermarkLogoSize", v[0])}
   690→                          disabled={disabled}
   691→                        />
   692→                      </div>
   693→                    </div>
   694→                  )}
   695→                </div>
   696→              )}
   697→            </div>
   698→          )}
   699→        </div>
   700→      </SectionCard>
   701→
   702→      {/* Timer + actions */}
   703→      <SectionCard title={t("recordingTimer")} icon={Clock} t={t}>
   704→        <div className="flex flex-col gap-4">
   705→          <div className="flex items-center justify-center gap-2.5 rounded-xl border border-border/50 bg-muted/30 py-4">
   706→            <Circle
   707→              className={cn(
   708→                "size-2.5",
   709→                rec.isRecording ? "fill-red-500 text-red-500 rec-dot" : "fill-muted-foreground/40 text-muted-foreground/40",
   710→              )}
   711→            />
   712→            <span className="font-mono text-3xl font-bold tabular-nums tracking-tight">
   713→              {rec.formatDuration(rec.elapsed)}
   714→            </span>
   715→          </div>
   716→
   717→          <div className="grid grid-cols-2 gap-2.5">
   718→            {rec.isRecording ? (
   719→              <Button
   720→                variant="outline"
   721→                onClick={rec.pauseRecording}
   722→                className="gap-2"
   723→                size="lg"
   724→              >
   725→                <Pause className="size-4" />
   726→                {t("pause")}
   727→              </Button>
   728→            ) : rec.isPaused ? (
   729→              <Button
   730→                variant="outline"
   731→                onClick={rec.resumeRecording}
   732→                className="gap-2"
   733→                size="lg"
   734→              >
   735→                <Play className="size-4" />
   736→                {t("resume")}
   737→              </Button>
   738→            ) : (
   739→              <Button
   740→                onClick={rec.startRecording}
   741→                disabled={!rec.canStart}
   742→                className="gap-2 bg-red-600 text-white hover:bg-red-600/90"
   743→                size="lg"
   744→              >
   745→                <Circle className="size-3.5 fill-current" />
   746→                {t("startRecording")}
   747→              </Button>
   748→            )}
   749→
   750→            {(rec.isRecording || rec.isPaused) && (
   751→              <Button
   752→                onClick={() => rec.stopRecording()}
   753→                variant="destructive"
   754→                className="gap-2"
   755→                size="lg"
   756→              >
   757→                <Square className="size-4 fill-current" />
   758→                {t("stop")}
   759→              </Button>
   760→            )}
   761→            {!(rec.isRecording || rec.isPaused) && (
   762→              <Button
   763→                onClick={rec.resetAll}
   764→                variant="outline"
   765→                className="gap-2"
   766→                size="lg"
   767→                disabled={rec.status === "idle" && !rec.result}
   768→              >
   769→                <RotateCcw className="size-4" />
   770→                {t("reset")}
   771→              </Button>
   772→            )}
   773→          </div>
   774→
   775→          {rec.features && !rec.features.getDisplayMedia ? (
   776→            <div className="flex items-start gap-2 rounded-lg border border-red-500/30 bg-red-500/5 px-3 py-2 text-[11px] text-red-600 dark:text-red-400">
   777→              <Info className="mt-0.5 size-3.5 shrink-0" />
   778→              <span>{t("errUnsupported")}</span>
   779→            </div>
   780→          ) : null}
   781→        </div>
   782→      </SectionCard>
   783→    </div>
   784→  );
   785→}
